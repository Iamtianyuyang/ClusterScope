--- scope-test script (extracted in_scope + probes) ---
FAILP=0
. /tmp/qa-verify/in_scope.sh
while IFS= read -r f; do [ -z "$f" ] && continue; in_scope "$f" || { echo "OUT-OF-SCOPE $f"; FAILP=$((FAILP+1)); }; done < /tmp/qa-verify/changed.txt
echo "OUT-COUNT=$FAILP"
echo "== negatives (must all print NEG-OUT) =="
for p in crates/storage/src/lib.rs crates/server/src/lib.rs Cargo.toml gauntlet.config.json .gauntlet/gauntlet.mjs; do in_scope "$p" && echo "NEG-MISSED $p" || echo "NEG-OUT $p"; done
echo "== positives (must all print POS-IN) =="
for p in crates/agent/src/config_loader.rs crates/agent/tests/no_root_agent_config.rs qa/harness/no-root-fixes-checks.sh gauntlet-tools/rust-gate.mjs deploy/clusterscope-agent.service; do in_scope "$p" && echo "POS-IN $p" || echo "POS-MISSED $p"; done
