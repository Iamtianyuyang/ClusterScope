#
#   mkdir -p ~/.config/systemd/user
#   cp deploy/clusterscope-agent.service ~/.config/systemd/user/
#   systemctl --user daemon-reload
#   systemctl --user enable --now clusterscope-agent
#
# %h expands to the installing user's home directory. The optional system-wide
# variant (needs root, writes into /etc and /usr/local) is deploy/agent.service.
#
# Requires a running user manager (systemd --user) and, to survive logout,
# logind linger for this user (`loginctl show-user $USER` / `Linger=yes`).
[Unit]
Description=ClusterScope GPU Agent (user)
After=network-online.target

[Service]
Type=simple
ExecStart=%h/.local/bin/clusterscope-agent -c %h/.config/clusterscope/agent.yaml
Restart=always
RestartSec=5

[Install]
WantedBy=default.target
